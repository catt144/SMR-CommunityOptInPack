    if not UIColony then return false, "no colony" end

    if state.held then

-- Owner, 2026-09-28: one switch that streams every train and station, instead of
-- selecting each one. Press to start, press again to stop; save, load and map change
-- stop it (re-press after an autosave). Read-only. Every `T.stream_cadence` game ms it
-- compares each train and each station row with the last tick and writes only what
-- changed as SMRTK_STREAM lines (file only, not the on-screen console). The first tick
-- writes everything, so a stream always starts from a full baseline.
T.stream_cadence = 1000
local function list(map, fmt)
    local keys = {}
    for k in pairs(map) do keys[#keys + 1] = k end
    table.sort(keys, function(a, b) return tostring(a) < tostring(b) end)
    local out = {}
    for _, k in ipairs(keys) do out[#out + 1] = fmt(k, map[k]) end
    return #out > 0 and table.concat(out, ",") or "none"
end

local function train_row(t)
    local cargo = {}
    for res, n in pairs(t.stockpiled_amount or empty_table) do
        if n and n > 0 then cargo[res] = n end
    end
    local assigned = {}
    for dest, by in pairs(t.assigned_resources or empty_table) do
        for res, n in pairs(by or empty_table) do
            if n and n > 0 then assigned[id(dest) .. ":" .. res] = n end
        end
    end
    local track = t.track
    local line = "none"
    if IsValid(track) then line = id(track:GetStartStation()) .. "-" .. id(track:GetEndStation()) end
    return { command = tostring(t.command), station = id(t.current_station),
        at_station = t.at_station and true or false, line = line,
        -- Actual interpolated movement, including acceleration/stops; not the nominal cap.
        -- 1.1.1.405907 CommonLua/LuaExportedDocs/Game/GameObject.lua:141-150,646.
        effective_speed = t:GetVelocity(),
        cargo = list(cargo, function(k, v) return k .. ":" .. v end),
        assigned = list(assigned, function(k, v) return k .. ":" .. v end),
        passengers = #(t.units or empty_table) }
end

local function stream_tick(state, d)
    local rows, changed = 0, 0
    for _, t in ipairs(UIColony.labels.Train or empty_table) do
        if IsValid(t) then
            local row = train_row(t)
            local key = id(t)
            local old = state.trains[key]
            local diff = not old
            for k, v in pairs(row) do diff = diff or old[k] ~= v end
            if diff then
                row.row, row.train = "train", key
                T.Log("STREAM", row, false)
                changed = changed + 1
            end
            state.trains[key] = row
            rows = rows + 1
        end
    end
    for _, s in ipairs(UIColony.labels.Station or empty_table) do
        if IsValid(s) and s.supply then
            local key = id(s)
            local last = state.stations[key] or {}
            state.stations[key] = last
            for _, res in ipairs(s.storable_resources or empty_table) do
                local sup = s.supply[res]
                if sup then
                    local stock = sup:GetActualAmount()
                    -- Brief 15: catch a visual-only loss even when stock did
                    -- not change. Count live cube objects, never repair them.
                    local cubes, expected = "none", "none"
                    if is_hub(s) and s.visual_cubes and s.visual_cubes[res] then
                        cubes = 0
                        for _, cube in pairs(s.visual_cubes[res]) do
                            if IsValid(cube) then cubes = cubes + 1 end
                        end
                        local cols = s.capacity_columns and s.capacity_columns[res]
                        if cols and s.max_z and s.max_storage_per_resource then
                            expected = s.visual_col_start and s.visual_col_start[res]
                                and Min(stock, s.max_storage_per_resource, cols * s.max_z * const.ResourceScale) // const.ResourceScale or 0
                        end
                    end
                    local old = last[res]
                    if not old or old.stock ~= stock or old.cubes ~= cubes or old.expected ~= expected then
                        T.Log("STREAM", { row = "stock", station = key, hub = is_hub(s), res = res,
                            stock = stock, before = old and old.stock or "start", cubes = cubes, expected_cubes = expected,
                            target = d and row_target(d, s, res) or "none",
                            calls = d and d.CallsFor(s) or "none" }, false)
                        last[res] = { stock = stock, cubes = cubes, expected = expected }
                        changed = changed + 1
                    end
                    rows = rows + 1
                end
            end
        end
    end
    state.ticks = state.ticks + 1
    return rows, changed
end

T.Bind(6, "Stream every train and station (press again to stop)", function(ctx)
    if not UIColony then return false, "no colony" end
    local state = ctx.state
    local d = rawget(_G, "SMROptInTrainDistribution")
    d = d and d.active and d or nil
    state.trains, state.stations, state.ticks, state.cancelled = {}, {}, 0, false
    state.mark = ctx.mark("STREAM_START")
    local rows, changed = stream_tick(state, d)
    state.thread = CreateGameTimeThread(function()
        while not state.cancelled do
            Sleep(T.stream_cadence)
            if state.cancelled then break end
            local ok, err = pcall(stream_tick, state, d)
            if not ok then
                T.Log("STREAM", { row = "error", error = tostring(err) }, false)
                state.cancelled = true
            end
        end
    end)
    return { mark = state.mark, cadence = T.stream_cadence, rows = rows, baseline = changed,
        distribution = d and true or false, game_time = GameTime() }
end, { mode = "armed", on_disarm = function(ctx)
    local state = ctx.state
    state.cancelled = true
    local thread = state.thread
    state.thread = nil
    if thread and IsValidThread(thread) and thread ~= CurrentThread() then DeleteThread(thread) end
    return { ticks = state.ticks, mark = ctx.mark("STREAM_STOP") }
end })

-- Why the 2026-09-18 witness could not prove a crossing: it inferred unloads by polling a train's
-- assigned_resources[hub] for a fall to zero. Vanilla Train:UnloadAll also unloads cargo assigned
-- to OTHER stations and cargo with no assignment at every stop (1.1.1.406343 Units/Train.lua:787-831),
-- so those unloads landed in other_in; trains on a route the arm-time table no longer matched were
-- skipped; and the rule ignored the hub's stock at arm.
--
-- This witness counts transfers where they happen. A transfer function is BRACKETED: every watched
-- station's stock is checkpointed on entry and exit, the entry gap is booked "other" and the
-- bracket's own delta to its category. Station stock writes by a train occur only in
-- LoadResourceForStation and UnloadAll's unload_cargo (Train.lua:744-831); a drone's in
-- DroneLoadResource/DroneUnloadResource (Station.lua:662-688, MultiResourceDepot.lua:201-215); the
-- cabin's only in the depot module's Tick (deliver/load_cabin). The wrappers stay installed and keep
-- recording while a run is disarmed, so a save, an autosave or Ultra cannot drop a transfer: a save
-- costs only polls, and stock moved by no bracket is booked "other", which can only make a proof
-- harder. Runtime install follows EF-058: every class whose lookup resolves to the method is wrapped,
-- and slot 7 proves it off live trains and stations.
--
-- The two crossings are different transfers (audit 2026-10-02 §2) and get separate verdicts:
--   hub route-to-route, per hub and resource: route B's trains loaded more at the hub than every
--     source that is not another route's train could supply:
--       out[B] > stock0 + other_in + drone_in + cabin_in + in[B] + in[unknown route]
--   depot map-to-map, per direction and resource, two links that must BOTH hold:
--     at the origin half, the cabin took more than non-train sources could supply:
--       cabin_out > stock0 + other_in + drone_in + cabin_in
--     at the destination half, its trains loaded more than non-cabin sources could supply:
--       sum(train_out) > stock0 + other_in + drone_in + sum(train_in)
-- Each excess is a lower bound on the crossed units. Those STRICT bounds need a near-empty station:
-- a stocked hub (about 2000 a resource, first attended check 2026-10-02) cannot clear them. Each
-- verdict therefore also has a NET form, from the conservation identity at the station,
--   out[B] - in[B] = sum over other routes of (in - out) + nontrain_net - (stock - stock0):
--   hub:   (out[B] - in[B]) - drawdown - max(nontrain_net, 0) - max(in[?] - out[?], 0) > 0
--   link1: (cabin_out - cabin_in) - drawdown - max(other_net + drone_net, 0) > 0
--   link2: (sum(train_out) - sum(train_in)) - drawdown - max(other_net + drone_net, 0) > 0
-- where drawdown = max(stock0 - stock, 0). A net excess is a lower bound on the OTHER routes' (or the
-- cabin's, or the trains') net delivery that the take matched: cargo crossed as a net flow. A verdict
-- names its bound. Controls: train_mismatch (a train's own cargo change against the station's) and
-- cabin_mismatch (cabin cargo against both halves) TAINT their station and resource, which no verdict
-- then uses; the first mismatches are logged in full. stuck counts brackets a poll found open.
-- Limit: "other" flows of both signs at one station inside one checkpoint gap (at most 250 game ms
-- with no bracketed event there) net against each other.
do
local T = SMRTK
-- Names to extend if brief 34's move renames them (a miss reports unwired, never a silent pass).
-- SMRElevatorDepot is brief 34's proposed name (TRAIN_MOVE_CHECKPOINT_20261002.md); the dev name stays.
local HUB_CLASSES = { "SMROptInTrainHubBase" }
local DEPOT_MODULES = { "SMRElevatorDepot", "SMRElevatorDepotDev" }
local RUNS = { "crossing_hub", "crossing_depot" }
local DEADLINE_HOURS = 12

local ledger = false          -- file-local: a save disarms a run, never this
local depth = 0               -- above 0 while a train or cabin bracket runs; nested transfers belong to it
local MISMATCH_LOG = 8
local route_keys = setmetatable({}, { __mode = "k" })
local wrapper_of = {}
local original_of = setmetatable({}, { __mode = "k" })

local function id(o)
    return IsValid(o) and (tostring(o.class) .. "(" .. tostring(o.handle) .. ")") or "none"
end
local function units(n)
    if type(n) ~= "number" then return tostring(n) end
    local scale = const.ResourceScale
    local sign, size = n < 0 and "-" or "", n < 0 and -n or n
    local whole, part = size // scale, size % scale
    if part == 0 then return sign .. tostring(whole) end
    return string.format("%s%d.%03d", sign, whole, part)
end
local function abs(n) return n < 0 and -n or n end
local function add(t, k, n) t[k] = (t[k] or 0) + n end
local function sum(t)
    local n = 0
    for _, v in pairs(t or empty_table) do n = n + v end
    return n
end
local function stock(st, res)
    local s = st.supply and st.supply[res]
    return s and s:GetActualAmount() or 0
end
local function depot_module()
    for _, name in ipairs(DEPOT_MODULES) do
        local d = rawget(_G, name)
        if type(d) == "table" and type(d.Tick) == "function" and type(d.PairOf) == "function" and d.CABIN then
            return d, name
        end
    end
end
local function is_hub(o)
    if not IsValid(o) then return false end
    for _, class in ipairs(HUB_CLASSES) do
        if IsKindOf(o, class) then return true end
    end
    return false
end
local function each_city(fn)
    for _, city in ipairs(rawget(_G, "Cities") or empty_table) do fn(city) end
end

-- ---- the books --------------------------------------------------------------------------------
local function route_key(train)
    local routes = train.city and train.city.train_track_routes
    local route = routes and train.track and routes[train.track]
    if type(route) ~= "table" then return "?" end
    local key = route_keys[route]
    if not key then
        local handles = {}
        for i = 1, #route do handles[i] = tostring(route[i].handle) end
        table.sort(handles)
        key = table.concat(handles, "-")
        route_keys[route] = key
    end
    if not ledger.routes[key] then
        local names = {}
        for i = 1, #route do names[i] = id(route[i]) end
        ledger.route_order[#ledger.route_order + 1] = key
        ledger.routes[key] = { label = "R" .. #ledger.route_order, path = table.concat(names, " > ") }
    end
    return key
end
local function label(key) return key == "?" and "R?" or ledger.routes[key].label end

local function new_book(st, kind)
    local b = { obj = st, kind = kind, stock0 = {}, last = {}, train_in = {}, train_out = {},
        drone_in = {}, drone_out = {}, cabin_in = {}, cabin_out = {}, other_in = {}, other_out = {},
        tainted = {}, events = { train = 0, drone = 0, cabin = 0 } }
    for _, res in ipairs(st.storable_resources or empty_table) do
        local n = stock(st, res)
        b.stock0[res], b.last[res] = n, n
    end
    return b
end
local function credit(b, res, cat, route, d)
    local size = abs(d)
    if cat == "train" then
        local side = d > 0 and b.train_in or b.train_out
        side[res] = side[res] or {}
        add(side[res], route, size)
    else
        add(b[cat .. (d > 0 and "_in" or "_out")], res, size)
    end
end
-- Books every change since the station's last checkpoint to `cat`; returns the deltas.
local function checkpoint(b, cat, route)
    local st, deltas = b.obj, {}
    if not IsValid(st) then return deltas end
    for _, res in ipairs(st.storable_resources or empty_table) do
        local now, was = stock(st, res), b.last[res]
        if was == nil then
            b.stock0[res] = now   -- a resource first seen mid-run counts as starting stock
        elseif now ~= was then
            credit(b, res, cat, route, now - was)
            deltas[res] = now - was
        end
        b.last[res] = now
    end
    return deltas
end

local function mismatch(kind, b, res, size, detail)
    if size == 0 then return end
    ledger[kind .. "_mismatch"] = ledger[kind .. "_mismatch"] + size
    b.tainted[res] = true
    if #ledger.mismatches < MISMATCH_LOG then
        detail.kind, detail.station, detail.resource, detail.size = kind, id(b.obj), res, units(size)
        detail.game_time = GameTime()
        ledger.mismatches[#ledger.mismatches + 1] = detail
    end
end
local function close() depth = depth > 0 and depth - 1 or 0 end

-- ---- the brackets -----------------------------------------------------------------------------
local function train_wrap(orig, name)
    return function(self, ...)
        local b = ledger and depth == 0 and ledger.books[self.current_station]
        if not b then return orig(self, ...) end
        depth = depth + 1
        checkpoint(b, "other")
        local before = {}
        for res, n in pairs(self.stockpiled_amount or empty_table) do before[res] = n end
        local result = table.pack(orig(self, ...))
        close()
        local route = route_key(self)
        local deltas = checkpoint(b, "train", route)
        local after, keys = self.stockpiled_amount or empty_table, {}
        for _, t in ipairs{ before, after, deltas } do for res in pairs(t) do keys[res] = true end end
        for res in pairs(keys) do
            local dt, ds = (after[res] or 0) - (before[res] or 0), deltas[res] or 0
            if dt + ds ~= 0 then
                mismatch("train", b, res, abs(dt + ds), { call = name, train = id(self), route = label(route),
                    train_delta = units(dt), station_delta = units(ds), station_stock = units(b.last[res] or 0),
                    capacity = units(b.obj.GetMaxStorage and b.obj:GetMaxStorage(res) or 0) })
            end
        end
        b.events.train = b.events.train + 1
        return table.unpack(result, 1, result.n)
    end
end
-- No depth here: a drone's load presentation sleeps (MultiResourceDepot.lua:192-199, 1.1.1.406343),
-- and a bracket held open across that yield is what read stuck=1 in the first attended check.
-- A train transfer during the sleep checkpoints the station itself, so nothing is booked twice.
local function drone_wrap(orig)
    return function(self, ...)
        local b = ledger and depth == 0 and ledger.books[self]
        if not b then return orig(self, ...) end
        checkpoint(b, "other")
        local result = table.pack(orig(self, ...))
        checkpoint(b, "drone")
        b.events.drone = b.events.drone + 1
        return table.unpack(result, 1, result.n)
    end
end
local function cabin_wrap(orig, d)
    return function(...)
        local s, u = d.PairOf()
        local bs = ledger and depth == 0 and s and ledger.books[s]
        local bu = bs and u and ledger.books[u]
        if not bu then return orig(...) end
        depth = depth + 1
        checkpoint(bs, "other"); checkpoint(bu, "other")
        local rec, before = rawget(s, d.CABIN), {}
        for res, n in pairs(type(rec) == "table" and rec.cargo or empty_table) do before[res] = n end
        local result = table.pack(orig(...))
        close()
        local ds, du = checkpoint(bs, "cabin"), checkpoint(bu, "cabin")
        rec = rawget(s, d.CABIN)
        local after, keys = type(rec) == "table" and rec.cargo or empty_table, {}
        for _, t in ipairs{ before, after, ds, du } do for res in pairs(t) do keys[res] = true end end
        for res in pairs(keys) do
            local size = abs((after[res] or 0) - (before[res] or 0) + (ds[res] or 0) + (du[res] or 0))
            if size ~= 0 then
                bu.tainted[res] = true
                mismatch("cabin", bs, res, size, { cabin_delta = units((after[res] or 0) - (before[res] or 0)),
                    surface_delta = units(ds[res] or 0), underground_delta = units(du[res] or 0) })
            end
        end
        if next(ds) or next(du) then
            bs.events.cabin = bs.events.cabin + 1
            bu.events.cabin = bu.events.cabin + 1
        end
        return table.unpack(result, 1, result.n)
    end
end

-- EF-058: wrap the method on EVERY class whose lookup resolves to it; idempotent.
local function install(base, method, make)
    local n = 0
    for name, class in pairs(g_Classes or empty_table) do
        if type(class) == "table" and (name == base or (class.__ancestors and class.__ancestors[base])) then
            local fn = class[method]
            if type(fn) == "function" then
                if not original_of[fn] then
                    local w = wrapper_of[fn]
                    if not w then w = make(fn, method); wrapper_of[fn] = w; original_of[w] = fn end
                    class[method] = w
                end
                n = n + 1
            end
        end
    end
    return n
end
local function wired(o, method) return original_of[o[method]] ~= nil end

local function watch(st, kind)
    if IsValid(st) and not ledger.books[st] then
        ledger.books[st] = new_book(st, kind)
        ledger.order[#ledger.order + 1] = st
    end
end
local function watch_all()
    each_city(function(city)
        for _, st in ipairs(city.labels and city.labels.Station or empty_table) do
            if is_hub(st) then watch(st, "hub") end
        end
    end)
    local d = depot_module()
    if d then
        local s, u = d.PairOf()
        if s then watch(s, "surface"); watch(u, "underground") end
    end
end

-- Installs (idempotent), watches new stations, and proves the wiring off live objects.
local function wire()
    local f = {
        classes_unload = install("Train", "UnloadAll", train_wrap),
        classes_load = install("Train", "LoadResourceForStation", train_wrap),
        classes_drone_unload = install("Station", "DroneUnloadResource", drone_wrap),
        classes_drone_load = install("Station", "DroneLoadResource", drone_wrap),
    }
    local d, name = depot_module()
    if d then
        if not original_of[d.Tick] then
            local w = cabin_wrap(d.Tick, d)
            original_of[w] = d.Tick
            d.Tick = w
        end
        f.depot_module = name
    else
        f.depot_module = "none"
    end
    f.cabin_wired = tostring(d and original_of[d.Tick] ~= nil or false)
    watch_all()
    local trains, trains_wired = 0, 0
    each_city(function(city)
        for _, t in ipairs(city.labels and city.labels.Train or empty_table) do
            if IsValid(t) then
                trains = trains + 1
                if wired(t, "UnloadAll") and wired(t, "LoadResourceForStation") then trains_wired = trains_wired + 1 end
            end
        end
    end)
    local stations, stations_wired, hubs, pair = 0, 0, 0, false
    for _, st in ipairs(ledger.order) do
        local b = ledger.books[st]
        if IsValid(st) then
            stations = stations + 1
            if b.kind == "hub" then hubs = hubs + 1 elseif b.kind == "surface" then pair = true end
            if wired(st, "DroneUnloadResource") and wired(st, "DroneLoadResource") then stations_wired = stations_wired + 1 end
        end
    end
    f.trains, f.trains_wired, f.watched, f.watched_wired, f.hubs = trains, trains_wired, stations, stations_wired, hubs
    f.depot_pair = tostring(pair)
    f.wiring_ok = tostring(trains_wired == trains and stations_wired == stations
        and (f.depot_module == "none" or f.cabin_wired == "true"))
    return f
end

-- ---- verdicts ---------------------------------------------------------------------------------
local function base_in(b, res)
    return (b.stock0[res] or 0) + (b.other_in[res] or 0) + (b.drone_in[res] or 0)
end
local function drawdown(b, res) return Max((b.stock0[res] or 0) - (b.last[res] or 0), 0) end
local function net(b, cat, res) return (b[cat .. "_in"][res] or 0) - (b[cat .. "_out"][res] or 0) end
-- The route with the best case; returns route, strict excess, net excess.
local function hub_excess(b, res)
    local best_route, best_strict, best_net
    local ins, outs = b.train_in[res] or empty_table, b.train_out[res] or empty_table
    local nontrain = Max(net(b, "other", res) + net(b, "drone", res) + net(b, "cabin", res), 0)
    local unknown = Max((ins["?"] or 0) - (outs["?"] or 0), 0)
    for route, out in pairs(outs) do
        if route ~= "?" then
            local own = ins[route] or 0
            local strict = out - base_in(b, res) - (b.cabin_in[res] or 0) - own - (ins["?"] or 0)
            local netx = out - own - drawdown(b, res) - nontrain - unknown
            if not best_route or Max(strict, netx) > Max(best_strict, best_net) then
                best_route, best_strict, best_net = route, strict, netx
            end
        end
    end
    return best_route, best_strict, best_net
end
local function noncabin(b, res) return Max(net(b, "other", res) + net(b, "drone", res), 0) end
local function cabin_took_train_cargo(b, res)   -- link 1, at the origin half: strict, net
    return (b.cabin_out[res] or 0) - base_in(b, res) - (b.cabin_in[res] or 0),
        -net(b, "cabin", res) - drawdown(b, res) - noncabin(b, res)
end
local function trains_took_cabin_cargo(b, res)  -- link 2, at the destination half: strict, net
    return sum(b.train_out[res]) - base_in(b, res) - sum(b.train_in[res]),
        sum(b.train_out[res]) - sum(b.train_in[res]) - drawdown(b, res) - noncabin(b, res)
end
local function bound(strict, netx) return strict > 0 and "strict" or (netx > 0 and "net" or nil) end
local function resources()
    local seen, list = {}, {}
    for _, st in ipairs(ledger.order) do
        for res in pairs(ledger.books[st].stock0) do
            if not seen[res] then seen[res] = true; list[#list + 1] = res end
        end
    end
    table.sort(list)
    return list
end
local function halves()
    local s, u
    for _, st in ipairs(ledger.order) do
        local b = ledger.books[st]
        if b.kind == "surface" then s = b elseif b.kind == "underground" then u = b end
    end
    return s, u
end
local function hub_proof()
    for _, st in ipairs(ledger.order) do
        local b = ledger.books[st]
        if b.kind == "hub" then
            for _, res in ipairs(resources()) do
                local route, strict, netx = hub_excess(b, res)
                local how = route and not b.tainted[res] and bound(strict, netx)
                if how then
                    return { crossing = "hub", hub = id(st), resource = res, route = label(route),
                        route_path = ledger.routes[route].path, bound = how,
                        crossed_at_least = units(how == "strict" and strict or netx),
                        strict_excess = units(strict), net_excess = units(netx) }
                end
            end
        end
    end
end
local function depot_proof()
    local s, u = halves()
    if not (s and u) then return end
    for _, res in ipairs(resources()) do
        for _, leg in ipairs{ { "down", s, u }, { "up", u, s } } do
            local s1, n1 = cabin_took_train_cargo(leg[2], res)
            local s2, n2 = trains_took_cabin_cargo(leg[3], res)
            local b1, b2 = bound(s1, n1), bound(s2, n2)
            if b1 and b2 and not leg[2].tainted[res] and not leg[3].tainted[res] then
                return { crossing = "depot_" .. leg[1], resource = res, origin = id(leg[2].obj), destination = id(leg[3].obj),
                    bound = (b1 == "strict" and b2 == "strict") and "strict" or "net",
                    cabin_took_train_cargo = units(b1 == "strict" and s1 or n1),
                    trains_took_cabin_cargo = units(b2 == "strict" and s2 or n2) }
            end
        end
    end
end
local function controls()
    local events, tainted = { train = 0, drone = 0, cabin = 0 }, 0
    for _, st in ipairs(ledger.order) do
        for k, n in pairs(ledger.books[st].events) do events[k] = events[k] + n end
        for _ in pairs(ledger.books[st].tainted) do tainted = tainted + 1 end
    end
    return { train_mismatch = units(ledger.train_mismatch), cabin_mismatch = units(ledger.cabin_mismatch), tainted = tainted,
        stuck = ledger.stuck, train_events = events.train, drone_events = events.drone, cabin_events = events.cabin,
        polls = ledger.polls, rearms = ledger.rearms, blind_ms = ledger.blind_ms,
        elapsed_ms = GameTime() - ledger.started, errors = T.error_count }
end
local function clean() return ledger.train_mismatch == 0 and ledger.cabin_mismatch == 0 and ledger.stuck == 0 end
local function poll()
    if depth ~= 0 then ledger.stuck = ledger.stuck + 1; depth = 0 end
    ledger.polls = ledger.polls + 1
    ledger.last_poll = GameTime()
    for _, st in ipairs(ledger.order) do checkpoint(ledger.books[st], "other") end
end
local function merge(into, from)
    for k, v in pairs(from or empty_table) do into[k] = v end
    return into
end

-- A loaded save replaces every object the books hold.
function OnMsg.LoadGame() ledger = false; depth = 0 end

local function cancel_runs()
    for _, key in ipairs(RUNS) do if T.armed[key] then T.Disarm(key, "replaced by a crossing slot") end end
end
for _, spec in ipairs{ { "crossing_hub", "Crossing: hub route-to-route proved", hub_proof },
        { "crossing_depot", "Crossing: depot map-to-map proved", depot_proof } } do
    local proof = spec[3]
    T.Trigger { id = spec[1], label = spec[2], cadence = 250,
        prepare = function(ctx)
            if not ledger then return false, "press slot 7 first: no ledger (or a save was loaded)" end
            local f = wire()
            if f.wiring_ok ~= "true" then return false, "wiring incomplete: " .. f.trains_wired .. "/" .. f.trains
                .. " trains, " .. f.watched_wired .. "/" .. f.watched .. " stations, cabin " .. f.cabin_wired end
            if ledger.runs > 0 then
                ledger.rearms = ledger.rearms + 1
                ledger.blind_ms = ledger.blind_ms + (GameTime() - ledger.last_poll)
            end
            ledger.runs = ledger.runs + 1
            local st = ctx.state
            st.deadline = GameTime() + DEADLINE_HOURS * const.HourDuration
            st.errors = T.error_count
            return merge(f, { rearms = ledger.rearms, blind_ms = ledger.blind_ms, deadline = st.deadline })
        end,
        when = function(ctx)
            if not ledger then return true, { verdict = "ledger_reset_by_load" } end
            poll()
            local f = controls()
            if T.error_count > ctx.state.errors then f.verdict = "new_error"; return true, f end
            local found = proof()
            if found then f.verdict = "proved"; return true, merge(f, found) end
            if GameTime() >= ctx.state.deadline then f.verdict = "deadline"; return true, f end
            return false
        end,
        ["do"] = { mark = true, pause = true, sound = true },
    }
end

local function run(key)
    cancel_runs()
    local ok, fields = T.Arm(key)
    if not ok then return false, fields.reason end
    local ran, why = T.Run("speed_ultra")
    if not ran then T.Disarm(key, "speed refused"); return false, why.reason end
    fields.trigger = key
    return fields
end

T.Bind(7, "Crossing witness: start the ledger + wiring proof (resets it)", function(ctx)
    local mark = ctx.mark("CROSSING_START")
    cancel_runs()
    ledger = { started = GameTime(), last_poll = GameTime(), polls = 0, runs = 0, rearms = 0, blind_ms = 0,
        books = {}, order = {}, routes = {}, route_order = {}, train_mismatch = 0, cabin_mismatch = 0, stuck = 0,
        mismatches = {} }
    depth = 0
    local f = wire()
    for _, st in ipairs(ledger.order) do
        local b = ledger.books[st]
        local stocked = {}
        for _, res in ipairs(resources()) do
            if (b.stock0[res] or 0) ~= 0 then stocked[#stocked + 1] = res .. "=" .. units(b.stock0[res]) end
        end
        ctx.log("DUMP", { leg = "crossing_watch", station = id(st), kind = b.kind,
            stock0 = #stocked > 0 and table.concat(stocked, " ") or "empty" })
    end
    f.mark = mark
    return f
end, { mode = "once" })
T.Bind(8, "Run until a hub route-to-route crossing is proved (Ultra; re-press after a save)",
    function() return run("crossing_hub") end, { mode = "once" })

local function install(base, method, make)

    for name, class in pairs(g_Classes or empty_table) do
        if type(class) == "table" and (name == base or (class.__ancestors and class.__ancestors[base])) then
            local fn = class[method]
            if type(fn) == "function" then
                if not original_of[fn] then
                    local w = wrapper_of[fn]
                    if not w then w = make(fn, method); wrapper_of[fn] = w; original_of[w] = fn end

                end
                n = n + 1
            end
        end
    end
    return n
end

-- ---------------------------------------------------------------------------
-- Fix 3: train cargo trap. Wrapped writers attribute every Lua write to a train's
-- stockpiled_amount; slot 3's poll reports changes nothing attributed.
-- ---------------------------------------------------------------------------
local trap = { on = false, installed = 0, bracketed = 0, unbracketed = 0, unattributed = 0, polls = 0, records = {} }
local TRAP_CAP = 60
local depth = setmetatable({}, { __mode = "k" })      -- train -> bracket depth
local accounted = setmetatable({}, { __mode = "k" })  -- train -> res -> delta since last poll
local snapshot = setmetatable({}, { __mode = "k" })   -- train -> res -> amount at last poll
local function note(train, res, delta)
    local acc = accounted[train] or {}
    accounted[train] = acc
    acc[res] = (acc[res] or 0) + delta
end
local function record(fields)
    if #trap.records < TRAP_CAP then trap.records[#trap.records + 1] = fields end
end
local function bracket_wrap(orig, method)
    return function(self, ...)
        if not trap.on or not IsKindOf(self, "Train") then return orig(self, ...) end
        depth[self] = (depth[self] or 0) + 1
        local before = {}
        for res, n in pairs(self.stockpiled_amount or empty_table) do before[res] = n end
        local result = table.pack(orig(self, ...))
        depth[self] = depth[self] - 1
        local after = self.stockpiled_amount or empty_table
        local keys = {}
        for res in pairs(before) do keys[res] = true end
        for res in pairs(after) do keys[res] = true end
        for res in pairs(keys) do
            local delta = (after[res] or 0) - (before[res] or 0)
            if delta ~= 0 then trap.bracketed = trap.bracketed + 1; note(self, res, delta) end
        end
        return table.unpack(result, 1, result.n)
    end
end
local function writer_wrap(orig, method)
    return function(self, ...)
        if not trap.on or not IsKindOf(self, "Train") or (depth[self] or 0) > 0 then return orig(self, ...) end
        local before = {}
        for res, n in pairs(self.stockpiled_amount or empty_table) do before[res] = n end
        local result = table.pack(orig(self, ...))
        local after = self.stockpiled_amount or empty_table
        local keys = {}
        for res in pairs(before) do keys[res] = true end
        for res in pairs(after) do keys[res] = true end
        for res in pairs(keys) do
            local delta = (after[res] or 0) - (before[res] or 0)
            if delta ~= 0 then
                trap.unbracketed = trap.unbracketed + 1
                note(self, res, delta)
                record({ leg = "unbracketed_write", train = id(self), method = method, resource = res,
                    before = units(before[res] or 0), after = units(after[res] or 0), delta = units(delta),
                    station = id(self.current_station), command = tostring(self.command), game_time = GameTime(),
                    stack = stack() })
            end
        end
        return table.unpack(result, 1, result.n)
    end
end
local function take_snapshot(train)
    local snap = {}
    for res, n in pairs(train.stockpiled_amount or empty_table) do snap[res] = n end
    snapshot[train] = snap
    accounted[train] = {}
end
local function poll()
    if not trap.on then return end
    trap.polls = trap.polls + 1
    each_train(function(train)
        local snap = snapshot[train]
        if not snap then take_snapshot(train) return end
        local acc = accounted[train] or empty_table
        local keys = {}
        for res in pairs(snap) do keys[res] = true end
        for res in pairs(train.stockpiled_amount or empty_table) do keys[res] = true end
        for res in pairs(keys) do
            local now_n = (train.stockpiled_amount or empty_table)[res] or 0
            local delta = now_n - (snap[res] or 0)
            if delta ~= (acc[res] or 0) then
                trap.unattributed = trap.unattributed + 1
                record({ leg = "unattributed_change", train = id(train), resource = res, delta = units(delta),
                    attributed = units(acc[res] or 0), before = units(snap[res] or 0), after = units(now_n),
                    station = id(train.current_station), command = tostring(train.command),
                    at_station = tostring(train.at_station), game_time = GameTime() })
            end
        end
        take_snapshot(train)
    end)
end

T.Bind(7, "Fix 3: train cargo trap on/off (log-only; re-press after a load)", function(ctx)
    if trap.on then
        trap.on = false
        return { trap = "off", bracketed_writes = trap.bracketed, unbracketed_writes = trap.unbracketed,
            unattributed_changes = trap.unattributed, polls = trap.polls, records = #trap.records }
    end
    local n = install("Train", "UnloadAll", bracket_wrap) + install("Train", "LoadResourceForStation", bracket_wrap)
    for _, method in ipairs({ "AddResource", "SetStoredAmount", "ClearAllResources", "ReturnStockpiledResources",
        "DroneLoadResource", "DroneUnloadResource" }) do
        n = n + install("MultiResourceCubeVisuals", method, writer_wrap)
    end
    trap.installed = n
    if n == 0 then return false, "no class resolves the train cargo writers" end
    trap.on = true
    trap.bracketed, trap.unbracketed, trap.unattributed, trap.polls, trap.records = 0, 0, 0, 0, {}
    local trains = 0
    each_train(function(train) take_snapshot(train); trains = trains + 1 end)
    local mark = ctx.mark("TRAP_ON_34B")
    return { trap = "on", classes = n, trains = trains, cap = TRAP_CAP, mark = mark,
        note = "slot 3's run polls every 500 game ms; press slot 8 to read" }
end, { mode = "once" })

local function train_fields(train, city)
    local st = train.current_station
    return { sitting = SITTING, row = "train", train = id(train), class = tostring(train.class),
        station = id(st), station_is_hub = tostring(is_hub(st)), at_spawn_track = tostring(train.at_spawn_track),
        at_station = tostring(train.at_station), command = tostring(train.command), game_time = GameTime() }
end
local function trains_summary(city)
    local total, in_hub = 0, 0
    each_train(function(train)
        total = total + 1
        if is_hub(train.current_station) and (train.at_spawn_track or train.at_station) then in_hub = in_hub + 1 end
    end)
    local b = bay()
    return { trains = total, parked_or_spawned_in_hub = in_hub, pool = ColonyGetPrefabs("Train", city),
        refused = b and b.stats and b.stats.refused or "bay not loaded", game_time = GameTime() }
end

T.Bind(4, "Fix 2: read trains, pool and refusals (paused)", function(ctx)
    if GetTimeFactor() ~= 0 then return false, "pause first" end
    local city = city_of(ctx.sel)
    local mark = ctx.mark("TRAINS_READ_34B")
    each_train(function(train, c) ctx.log("DUMP", train_fields(train, c)) end)
    local o = ctx.sel
    if is_hub(o) then
        ctx.log("DUMP", { sitting = SITTING, row = "hub", hub = id(o), trains_in_construction = o.trains_in_construction or 0,
            construction_queue = #(o.construction_queue or empty_table), interaction_mode = tostring(o.interaction_mode) })
    end
    local out = trains_summary(city)
    out.mark = mark
    return out
end, { mode = "once" })

T.Bind(5, "Fix 2: refusal control - vanilla AssignTrain on the selected hub (the guard must refuse)", function(ctx)
    if GetTimeFactor() ~= 0 then return false, "pause first" end
    local hub = ctx.sel
    if not is_hub(hub) then return false, "select the hub" end
    local track
    hub:ForEachConnectedTrack(function(t) track = t; return "break" end)
    if not IsValid(track) then return false, "the hub has no connected track reaching another station" end
    local city = city_of(hub)
    local before = trains_summary(city)
    local mark = ctx.mark("REFUSAL_CONTROL_34B")
    track:AssignTrain(hub)
    local after = trains_summary(city)
    return { mark = mark, hub = id(hub), track = id(track), trains_before = before.trains, trains_after = after.trains,
        pool_before = before.pool, pool_after = after.pool, refused_before = before.refused, refused_after = after.refused,
        verdict = (after.trains == before.trains and after.pool == before.pool and after.refused == before.refused + 1)
            and "refused" or "NOT_REFUSED" }
end, { mode = "once", needs = "selected" })

T.Bind(6, "Fix 2: add one Train prefab to the pool (only when it reads 0)", function(ctx)

    if pool > 0 then return false, "the pool already holds " .. tostring(pool) .. " train(s)" end
    local mark = ctx.mark("POOL_TOPUP_34B")
    ColonyAddPrefabs("Train", 1, nil, city)
    return { mark = mark, pool_before = pool, pool_after = ColonyGetPrefabs("Train", city) }
end, { mode = "once" })

    for n=1,10 do
        local entry = group[n]
        local label = entry and entry.label or "Unused in this group"
        local fn = entry and entry.fn or function() return false,"unused in this group" end
        local def, why = T.Bind(n, group.name .. ": " .. label, fn, entry and entry.opts or {mode="once"})

    group_index = index
