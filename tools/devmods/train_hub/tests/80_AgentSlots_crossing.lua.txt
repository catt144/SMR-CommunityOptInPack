-- Brief 33: the crossing witness, slots 7-10. Append after the depot slots; binds nothing else.
-- Report and predictions: OptInPack docs/agent/reports/TRAIN_CROSSING_WITNESS_20261002.md.
-- Nothing arms or mutates game state at load. Slot 7 installs pass-through wrappers (below) and
-- starts the ledger; 8 and 9 run game time at Ultra until their trigger pauses; 10 is a read.
--
-- Why the 2026-09-18 witness could not prove a crossing: it inferred unloads by polling a train's
-- assigned_resources[hub] for a fall to zero. Vanilla Train:UnloadAll also unloads cargo assigned
-- to OTHER stations and cargo with no assignment at every stop (1.1.1.406343 Units/Train.lua:787-831),
-- so those unloads landed in other_in; trains on a route the arm-time table no longer matched were
-- skipped; and the rule ignored the hub's stock at arm.
--
-- This witness counts transfers where they happen. A transfer function is BRACKETED: every watched
-- station's stock is checkpointed on entry and exit, the entry gap is booked "other" and the
-- bracket's own delta to its category. Station stock writes by a train occur only in
-- LoadResourceForStation and UnloadAll's unload_cargo (Train.lua:744-831); a drone's in
-- DroneLoadResource/DroneUnloadResource (Station.lua:662-688, MultiResourceDepot.lua:201-215); the
-- cabin's only in the depot module's Tick (deliver/load_cabin). The wrappers stay installed and keep
-- recording while a run is disarmed, so a save, an autosave or Ultra cannot drop a transfer: a save
-- costs only polls, and stock moved by no bracket is booked "other", which can only make a proof
-- harder. Runtime install follows EF-058: every class whose lookup resolves to the method is wrapped,
-- and slot 7 proves it off live trains and stations.
--
-- The two crossings are different transfers (audit 2026-10-02 §2) and get separate verdicts:
--   hub route-to-route, per hub and resource: route B's trains loaded more at the hub than every
--     source that is not another route's train could supply:
--       out[B] > stock0 + other_in + drone_in + cabin_in + in[B] + in[unknown route]
--   depot map-to-map, per direction and resource, two links that must BOTH hold:
--     at the origin half, the cabin took more than non-train sources could supply:
--       cabin_out > stock0 + other_in + drone_in + cabin_in
--     at the destination half, its trains loaded more than non-cabin sources could supply:
--       sum(train_out) > stock0 + other_in + drone_in + sum(train_in)
-- Each excess is a lower bound on the crossed units. Controls: train_mismatch (a train's own cargo
-- change against the station's), cabin_mismatch (cabin cargo against both halves), stuck brackets,
-- and wiring. A broken control ends a run with verdict control_broken, never a proof.
-- Limit: "other" flows of both signs at one station inside one checkpoint gap (at most 250 game ms
-- with no bracketed event there) net against each other.
do
local T = SMRTK
-- Names to extend if brief 34's move renames them (a miss reports unwired, never a silent pass).
-- SMRElevatorDepot is brief 34's proposed name (TRAIN_MOVE_CHECKPOINT_20261002.md); the dev name stays.
local HUB_CLASSES = { "SMROptInTrainHubBase" }
local DEPOT_MODULES = { "SMRElevatorDepot", "SMRElevatorDepotDev" }
local RUNS = { "crossing_hub", "crossing_depot" }
local DEADLINE_HOURS = 12

local ledger = false          -- file-local: a save disarms a run, never this
local depth = 0               -- above 0 while a bracket runs; nested transfers belong to it
local route_keys = setmetatable({}, { __mode = "k" })
local wrapper_of = {}
local original_of = setmetatable({}, { __mode = "k" })

local function id(o)
    return IsValid(o) and (tostring(o.class) .. "(" .. tostring(o.handle) .. ")") or "none"
end
local function units(n)
    if type(n) ~= "number" then return tostring(n) end
    local scale = const.ResourceScale
    local sign, size = n < 0 and "-" or "", n < 0 and -n or n
    local whole, part = size // scale, size % scale
    if part == 0 then return sign .. tostring(whole) end
    return string.format("%s%d.%03d", sign, whole, part)
end
local function abs(n) return n < 0 and -n or n end
local function add(t, k, n) t[k] = (t[k] or 0) + n end
local function sum(t)
    local n = 0
    for _, v in pairs(t or empty_table) do n = n + v end
    return n
end
local function stock(st, res)
    local s = st.supply and st.supply[res]
    return s and s:GetActualAmount() or 0
end
local function depot_module()
    for _, name in ipairs(DEPOT_MODULES) do
        local d = rawget(_G, name)
        if type(d) == "table" and type(d.Tick) == "function" and type(d.PairOf) == "function" and d.CABIN then
            return d, name
        end
    end
end
local function is_hub(o)
    if not IsValid(o) then return false end
    for _, class in ipairs(HUB_CLASSES) do
        if IsKindOf(o, class) then return true end
    end
    return false
end
local function each_city(fn)
    for _, city in ipairs(rawget(_G, "Cities") or empty_table) do fn(city) end
end

-- ---- the books --------------------------------------------------------------------------------
local function route_key(train)
    local routes = train.city and train.city.train_track_routes
    local route = routes and train.track and routes[train.track]
    if type(route) ~= "table" then return "?" end
    local key = route_keys[route]
    if not key then
        local handles = {}
        for i = 1, #route do handles[i] = tostring(route[i].handle) end
        table.sort(handles)
        key = table.concat(handles, "-")
        route_keys[route] = key
    end
    if not ledger.routes[key] then
        local names = {}
        for i = 1, #route do names[i] = id(route[i]) end
        ledger.route_order[#ledger.route_order + 1] = key
        ledger.routes[key] = { label = "R" .. #ledger.route_order, path = table.concat(names, " > ") }
    end
    return key
end
local function label(key) return key == "?" and "R?" or ledger.routes[key].label end

local function new_book(st, kind)
    local b = { obj = st, kind = kind, stock0 = {}, last = {}, train_in = {}, train_out = {},
        drone_in = {}, drone_out = {}, cabin_in = {}, cabin_out = {}, other_in = {}, other_out = {},
        events = { train = 0, drone = 0, cabin = 0 } }
    for _, res in ipairs(st.storable_resources or empty_table) do
        local n = stock(st, res)
        b.stock0[res], b.last[res] = n, n
    end
    return b
end
local function credit(b, res, cat, route, d)
    local size = abs(d)
    if cat == "train" then
        local side = d > 0 and b.train_in or b.train_out
        side[res] = side[res] or {}
        add(side[res], route, size)
    else
        add(b[cat .. (d > 0 and "_in" or "_out")], res, size)
    end
end
-- Books every change since the station's last checkpoint to `cat`; returns the deltas.
local function checkpoint(b, cat, route)
    local st, deltas = b.obj, {}
    if not IsValid(st) then return deltas end
    for _, res in ipairs(st.storable_resources or empty_table) do
        local now, was = stock(st, res), b.last[res]
        if was == nil then
            b.stock0[res] = now   -- a resource first seen mid-run counts as starting stock
        elseif now ~= was then
            credit(b, res, cat, route, now - was)
            deltas[res] = now - was
        end
        b.last[res] = now
    end
    return deltas
end

-- ---- the brackets -----------------------------------------------------------------------------
local function train_wrap(orig)
    return function(self, ...)
        local b = ledger and depth == 0 and ledger.books[self.current_station]
        if not b then return orig(self, ...) end
        depth = depth + 1
        checkpoint(b, "other")
        local before = {}
        for res, n in pairs(self.stockpiled_amount or empty_table) do before[res] = n end
        local result = table.pack(orig(self, ...))
        depth = depth - 1
        local deltas = checkpoint(b, "train", route_key(self))
        local after, seen = self.stockpiled_amount or empty_table, {}
        for res, n in pairs(after) do
            seen[res] = true
            ledger.train_mismatch = ledger.train_mismatch + abs(n - (before[res] or 0) + (deltas[res] or 0))
        end
        for res, n in pairs(before) do
            if not seen[res] then seen[res] = true; ledger.train_mismatch = ledger.train_mismatch + abs((deltas[res] or 0) - n) end
        end
        for res, d in pairs(deltas) do
            if not seen[res] then ledger.train_mismatch = ledger.train_mismatch + abs(d) end
        end
        b.events.train = b.events.train + 1
        return table.unpack(result, 1, result.n)
    end
end
local function drone_wrap(orig)
    return function(self, ...)
        local b = ledger and depth == 0 and ledger.books[self]
        if not b then return orig(self, ...) end
        depth = depth + 1
        checkpoint(b, "other")
        local result = table.pack(orig(self, ...))
        depth = depth - 1
        checkpoint(b, "drone")
        b.events.drone = b.events.drone + 1
        return table.unpack(result, 1, result.n)
    end
end
local function cabin_wrap(orig, d)
    return function(...)
        local s, u = d.PairOf()
        local bs = ledger and depth == 0 and s and ledger.books[s]
        local bu = bs and u and ledger.books[u]
        if not bu then return orig(...) end
        depth = depth + 1
        checkpoint(bs, "other"); checkpoint(bu, "other")
        local rec, before = rawget(s, d.CABIN), {}
        for res, n in pairs(type(rec) == "table" and rec.cargo or empty_table) do before[res] = n end
        local result = table.pack(orig(...))
        depth = depth - 1
        local ds, du = checkpoint(bs, "cabin"), checkpoint(bu, "cabin")
        rec = rawget(s, d.CABIN)
        local after, keys = type(rec) == "table" and rec.cargo or empty_table, {}
        for _, t in ipairs{ before, after, ds, du } do for res in pairs(t) do keys[res] = true end end
        for res in pairs(keys) do
            ledger.cabin_mismatch = ledger.cabin_mismatch
                + abs((after[res] or 0) - (before[res] or 0) + (ds[res] or 0) + (du[res] or 0))
        end
        if next(ds) or next(du) then
            bs.events.cabin = bs.events.cabin + 1
            bu.events.cabin = bu.events.cabin + 1
        end
        return table.unpack(result, 1, result.n)
    end
end

-- EF-058: wrap the method on EVERY class whose lookup resolves to it; idempotent.
local function install(base, method, make)
    local n = 0
    for name, class in pairs(g_Classes or empty_table) do
        if type(class) == "table" and (name == base or (class.__ancestors and class.__ancestors[base])) then
            local fn = class[method]
            if type(fn) == "function" then
                if not original_of[fn] then
                    local w = wrapper_of[fn]
                    if not w then w = make(fn); wrapper_of[fn] = w; original_of[w] = fn end
                    class[method] = w
                end
                n = n + 1
            end
        end
    end
    return n
end
local function wired(o, method) return original_of[o[method]] ~= nil end

local function watch(st, kind)
    if IsValid(st) and not ledger.books[st] then
        ledger.books[st] = new_book(st, kind)
        ledger.order[#ledger.order + 1] = st
    end
end
local function watch_all()
    each_city(function(city)
        for _, st in ipairs(city.labels and city.labels.Station or empty_table) do
            if is_hub(st) then watch(st, "hub") end
        end
    end)
    local d = depot_module()
    if d then
        local s, u = d.PairOf()
        if s then watch(s, "surface"); watch(u, "underground") end
    end
end

-- Installs (idempotent), watches new stations, and proves the wiring off live objects.
local function wire()
    local f = {
        classes_unload = install("Train", "UnloadAll", train_wrap),
        classes_load = install("Train", "LoadResourceForStation", train_wrap),
        classes_drone_unload = install("Station", "DroneUnloadResource", drone_wrap),
        classes_drone_load = install("Station", "DroneLoadResource", drone_wrap),
    }
    local d, name = depot_module()
    if d then
        if not original_of[d.Tick] then
            local w = cabin_wrap(d.Tick, d)
            original_of[w] = d.Tick
            d.Tick = w
        end
        f.depot_module = name
    else
        f.depot_module = "none"
    end
    f.cabin_wired = tostring(d and original_of[d.Tick] ~= nil or false)
    watch_all()
    local trains, trains_wired = 0, 0
    each_city(function(city)
        for _, t in ipairs(city.labels and city.labels.Train or empty_table) do
            if IsValid(t) then
                trains = trains + 1
                if wired(t, "UnloadAll") and wired(t, "LoadResourceForStation") then trains_wired = trains_wired + 1 end
            end
        end
    end)
    local stations, stations_wired, hubs, pair = 0, 0, 0, false
    for _, st in ipairs(ledger.order) do
        local b = ledger.books[st]
        if IsValid(st) then
            stations = stations + 1
            if b.kind == "hub" then hubs = hubs + 1 elseif b.kind == "surface" then pair = true end
            if wired(st, "DroneUnloadResource") and wired(st, "DroneLoadResource") then stations_wired = stations_wired + 1 end
        end
    end
    f.trains, f.trains_wired, f.watched, f.watched_wired, f.hubs = trains, trains_wired, stations, stations_wired, hubs
    f.depot_pair = tostring(pair)
    f.wiring_ok = tostring(trains_wired == trains and stations_wired == stations
        and (f.depot_module == "none" or f.cabin_wired == "true"))
    return f
end

-- ---- verdicts ---------------------------------------------------------------------------------
local function base_in(b, res)
    return (b.stock0[res] or 0) + (b.other_in[res] or 0) + (b.drone_in[res] or 0)
end
local function hub_excess(b, res)
    local best, best_route
    local ins = b.train_in[res] or empty_table
    for route, out in pairs(b.train_out[res] or empty_table) do
        if route ~= "?" then
            local excess = out - base_in(b, res) - (b.cabin_in[res] or 0) - (ins[route] or 0) - (ins["?"] or 0)
            if not best or excess > best then best, best_route = excess, route end
        end
    end
    return best_route, best
end
local function cabin_took_train_cargo(b, res)   -- link 1, at the origin half
    return (b.cabin_out[res] or 0) - base_in(b, res) - (b.cabin_in[res] or 0)
end
local function trains_took_cabin_cargo(b, res)  -- link 2, at the destination half
    return sum(b.train_out[res]) - base_in(b, res) - sum(b.train_in[res])
end
local function resources()
    local seen, list = {}, {}
    for _, st in ipairs(ledger.order) do
        for res in pairs(ledger.books[st].stock0) do
            if not seen[res] then seen[res] = true; list[#list + 1] = res end
        end
    end
    table.sort(list)
    return list
end
local function halves()
    local s, u
    for _, st in ipairs(ledger.order) do
        local b = ledger.books[st]
        if b.kind == "surface" then s = b elseif b.kind == "underground" then u = b end
    end
    return s, u
end
local function hub_proof()
    for _, st in ipairs(ledger.order) do
        local b = ledger.books[st]
        if b.kind == "hub" then
            for _, res in ipairs(resources()) do
                local route, excess = hub_excess(b, res)
                if route and excess > 0 then
                    return { crossing = "hub", hub = id(st), resource = res, route = label(route),
                        route_path = ledger.routes[route].path, crossed_at_least = units(excess) }
                end
            end
        end
    end
end
local function depot_proof()
    local s, u = halves()
    if not (s and u) then return end
    for _, res in ipairs(resources()) do
        for _, leg in ipairs{ { "down", s, u }, { "up", u, s } } do
            local e1, e2 = cabin_took_train_cargo(leg[2], res), trains_took_cabin_cargo(leg[3], res)
            if e1 > 0 and e2 > 0 then
                return { crossing = "depot_" .. leg[1], resource = res, origin = id(leg[2].obj), destination = id(leg[3].obj),
                    cabin_took_train_cargo = units(e1), trains_took_cabin_cargo = units(e2) }
            end
        end
    end
end
local function controls()
    local events = { train = 0, drone = 0, cabin = 0 }
    for _, st in ipairs(ledger.order) do
        for k, n in pairs(ledger.books[st].events) do events[k] = events[k] + n end
    end
    return { train_mismatch = units(ledger.train_mismatch), cabin_mismatch = units(ledger.cabin_mismatch),
        stuck = ledger.stuck, train_events = events.train, drone_events = events.drone, cabin_events = events.cabin,
        polls = ledger.polls, rearms = ledger.rearms, blind_ms = ledger.blind_ms,
        elapsed_ms = GameTime() - ledger.started, errors = T.error_count }
end
local function broken() return ledger.train_mismatch ~= 0 or ledger.cabin_mismatch ~= 0 or ledger.stuck ~= 0 end
local function poll()
    if depth ~= 0 then ledger.stuck = ledger.stuck + 1; depth = 0 end
    ledger.polls = ledger.polls + 1
    ledger.last_poll = GameTime()
    for _, st in ipairs(ledger.order) do checkpoint(ledger.books[st], "other") end
end
local function merge(into, from)
    for k, v in pairs(from or empty_table) do into[k] = v end
    return into
end

-- A loaded save replaces every object the books hold.
function OnMsg.LoadGame() ledger = false; depth = 0 end

local function cancel_runs()
    for _, key in ipairs(RUNS) do if T.armed[key] then T.Disarm(key, "replaced by a crossing slot") end end
end
for _, spec in ipairs{ { "crossing_hub", "Crossing: hub route-to-route proved", hub_proof },
        { "crossing_depot", "Crossing: depot map-to-map proved", depot_proof } } do
    local proof = spec[3]
    T.Trigger { id = spec[1], label = spec[2], cadence = 250,
        prepare = function(ctx)
            if not ledger then return false, "press slot 7 first: no ledger (or a save was loaded)" end
            local f = wire()
            if f.wiring_ok ~= "true" then return false, "wiring incomplete: " .. f.trains_wired .. "/" .. f.trains
                .. " trains, " .. f.watched_wired .. "/" .. f.watched .. " stations, cabin " .. f.cabin_wired end
            if ledger.runs > 0 then
                ledger.rearms = ledger.rearms + 1
                ledger.blind_ms = ledger.blind_ms + (GameTime() - ledger.last_poll)
            end
            ledger.runs = ledger.runs + 1
            local st = ctx.state
            st.deadline = GameTime() + DEADLINE_HOURS * const.HourDuration
            st.errors = T.error_count
            return merge(f, { rearms = ledger.rearms, blind_ms = ledger.blind_ms, deadline = st.deadline })
        end,
        when = function(ctx)
            if not ledger then return true, { verdict = "ledger_reset_by_load" } end
            poll()
            local f = controls()
            if T.error_count > ctx.state.errors then f.verdict = "new_error"; return true, f end
            if broken() then f.verdict = "control_broken"; return true, f end
            local found = proof()
            if found then f.verdict = "proved"; return true, merge(f, found) end
            if GameTime() >= ctx.state.deadline then f.verdict = "deadline"; return true, f end
            return false
        end,
        ["do"] = { mark = true, pause = true, sound = true },
    }
end

local function run(key)
    cancel_runs()
    local ok, fields = T.Arm(key)
    if not ok then return false, fields.reason end
    local ran, why = T.Run("speed_ultra")
    if not ran then T.Disarm(key, "speed refused"); return false, why.reason end
    fields.trigger = key
    return fields
end

T.Bind(7, "Crossing witness: start the ledger + wiring proof (resets it)", function(ctx)
    local mark = ctx.mark("CROSSING_START")
    cancel_runs()
    ledger = { started = GameTime(), last_poll = GameTime(), polls = 0, runs = 0, rearms = 0, blind_ms = 0,
        books = {}, order = {}, routes = {}, route_order = {}, train_mismatch = 0, cabin_mismatch = 0, stuck = 0 }
    depth = 0
    local f = wire()
    for _, st in ipairs(ledger.order) do
        local b = ledger.books[st]
        local stocked = {}
        for _, res in ipairs(resources()) do
            if (b.stock0[res] or 0) ~= 0 then stocked[#stocked + 1] = res .. "=" .. units(b.stock0[res]) end
        end
        ctx.log("DUMP", { leg = "crossing_watch", station = id(st), kind = b.kind,
            stock0 = #stocked > 0 and table.concat(stocked, " ") or "empty" })
    end
    f.mark = mark
    return f
end, { mode = "once" })
T.Bind(8, "Run until a hub route-to-route crossing is proved (Ultra; re-press after a save)",
    function() return run("crossing_hub") end, { mode = "once" })
T.Bind(9, "Run until a depot map-to-map crossing is proved (Ultra; re-press after a save)",
    function() return run("crossing_depot") end, { mode = "once" })

T.Bind(10, "Read the crossing ledger and both verdicts (pause first)", function(ctx)
    if GetTimeFactor() ~= 0 then return false, "pause first" end
    if not ledger then return false, "no ledger: press slot 7" end
    local mark = ctx.mark("CROSSING_READ")
    poll()
    for _, key in ipairs(ledger.route_order) do
        ctx.log("DUMP", { leg = "crossing_route", route = ledger.routes[key].label, path = ledger.routes[key].path })
    end
    local s, u = halves()
    for _, st in ipairs(ledger.order) do
        local b = ledger.books[st]
        for _, res in ipairs(resources()) do
            local active = (b.last[res] or 0) ~= (b.stock0[res] or 0) or next(b.train_in[res] or empty_table)
                or next(b.train_out[res] or empty_table)
            for _, cat in ipairs{ "drone_in", "drone_out", "cabin_in", "cabin_out", "other_in", "other_out" } do
                active = active or (b[cat][res] or 0) ~= 0
            end
            if active then
                local ins, outs = {}, {}
                for route, n in pairs(b.train_in[res] or empty_table) do ins[#ins + 1] = label(route) .. "=" .. units(n) end
                for route, n in pairs(b.train_out[res] or empty_table) do outs[#outs + 1] = label(route) .. "=" .. units(n) end
                table.sort(ins); table.sort(outs)
                local row = { leg = "crossing_ledger", station = id(st), kind = b.kind, resource = res,
                    stock0 = units(b.stock0[res] or 0), stock = units(b.last[res] or 0),
                    train_in = #ins > 0 and table.concat(ins, " ") or "none",
                    train_out = #outs > 0 and table.concat(outs, " ") or "none" }
                for _, cat in ipairs{ "drone_in", "drone_out", "cabin_in", "cabin_out", "other_in", "other_out" } do
                    row[cat] = units(b[cat][res] or 0)
                end
                if b.kind == "hub" then
                    local route, excess = hub_excess(b, res)
                    row.hub_route, row.hub_excess = route and label(route) or "none", route and units(excess) or "none"
                else
                    row.cabin_took_train_cargo = units(cabin_took_train_cargo(b, res))
                    row.trains_took_cabin_cargo = units(trains_took_cabin_cargo(b, res))
                end
                ctx.log("DUMP", row)
            end
        end
    end
    local f = controls()
    f.mark = mark
    local hub, depot = hub_proof(), depot_proof()
    f.hub_verdict = hub and string.format("proved %s %s %s crossed>=%s", hub.hub, hub.resource, hub.route,
        hub.crossed_at_least) or "none"
    f.depot_verdict = depot and string.format("proved %s %s link1=%s link2=%s", depot.crossing, depot.resource,
        depot.cabin_took_train_cargo, depot.trains_took_cabin_cargo) or "none"
    f.depot_pair = tostring(s ~= nil and u ~= nil)
    f.controls_ok = tostring(not broken())
    return f
end, { mode = "once" })
end
